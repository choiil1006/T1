; LS ELECTRIC XGB XBC-DR32H ladder (XG5000, mnemonic / IL view)
; CPU type in XG5000: XGB-XBCH
; Inputs  (DC24V, 16 pts): P00000~P0000F   Outputs (relay, 16 pts): P00040~P0004F
;
; Inputs (from Pico through the opto module)
;   P00000 pad 1 DL (Pico GP12)   P00001 pad 2 UL (GP13)   P00002 pad 3 CN (GP14)
;   P00003 pad 4 UR (GP15)        P00004 pad 5 DR (GP16)
;   P00005 score bit0 (GP17)      P00006 score bit1 (GP18) P00007 score bit2 (GP19)
;   P00008 combo on   (GP20)
; Outputs
;   P00040~P00044 pad 1~5 blue
;   P00045 combo yellow - red channel of all 5 strips (parallel)
;   P00046 combo yellow - green channel of all 5 strips (parallel)
;   P00047~P0004B score lamps 1~5 (level N lights N lamps)
;   spare: inputs P00009~P0000F, outputs P0004C~P0004F

; --- pad blue ---
LOAD  P00000
OUT   P00040
LOAD  P00001
OUT   P00041
LOAD  P00002
OUT   P00042
LOAD  P00003
OUT   P00043
LOAD  P00004
OUT   P00044

; --- combo yellow (red + green on every pad) ---
LOAD  P00008
OUT   P00045
OUT   P00046

; --- score lamps: level = bit0*1 + bit1*2 + bit2*4 (0~5) ---
LOAD  P00005
OR    P00006
OR    P00007
OUT   P00047        ; level >= 1
LOAD  P00006
OR    P00007
OUT   P00048        ; level >= 2
LOAD  P00005
AND   P00006
OR    P00007
OUT   P00049        ; level >= 3
LOAD  P00007
OUT   P0004A        ; level >= 4
LOAD  P00005
OR    P00006
AND   P00007
OUT   P0004B        ; level >= 5
END
